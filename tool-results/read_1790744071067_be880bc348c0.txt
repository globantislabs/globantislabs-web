    45→    45→export async function generateMetadata({
    46→    46→  params,
    47→    47→}: {

    49→    49→}): Promise<Metadata> {
    50→    50→  const { slug } = await params;
    51→    51→  const ind = industries.find((i) => i.slug === slug);
    52→    52→  if (!ind) return {};
    53→    53→  const title = `${ind.title} | Globantis Labs`;
    54→    54→  const description =
    55→    55→    ind.intro ??
    56→    56→    ind.heroHeading ??
    57→    57→    ind.sections?.[0]?.paragraphs?.[0] ??
    58→    58→    ind.tagline ??
    59→    59→    ind.title;
    60→    60→  return buildMetadata({
    61→    61→    title,
    62→    62→    description,
    63→    63→    path: `/industries/${slug}`,
    64→    64→    keywords: [ind.title, ind.tagline, ...(ind.focusAreas ?? [])].filter(
    65→    65→      Boolean
