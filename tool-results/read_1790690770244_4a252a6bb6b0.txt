    46→export async function generateMetadata({
    47→  params,
    48→}: {
    49→  params: Promise<{ slug: string }>;
    50→}): Promise<Metadata> {
    51→  const { slug } = await params;
    52→  const ind = industries.find((i) => i.slug === slug);
    53→  if (!ind) return {};
    54→  const title = `${ind.title} | Globantis Labs`;
    55→  const description =
    56→    ind.intro ??
    57→    ind.heroHeading ??
    58→    ind.sections?.[0]?.paragraphs?.[0] ??
    59→    ind.tagline ??
    60→    ind.title;
    61→  return buildMetadata({
    62→    title,
    63→    description,
    64→    path: `/industries/${slug}`,
    65→    keywords: [ind.title, ind.tagline, ...(ind.focusAreas ?? [])].filter(
    66→      Boolean
    67→    ) as string[],
    68→  });
    69→}
