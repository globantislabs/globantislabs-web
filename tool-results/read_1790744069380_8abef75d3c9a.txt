    45→export async function generateMetadata({
    46→  params,
    47→}: {
    48→  params: Promise<{ slug: string }>;
    49→}): Promise<Metadata> {
    50→  const { slug } = await params;
    51→  const ind = industries.find((i) => i.slug === slug);
    52→  if (!ind) return {};
    53→  const title = `${ind.title} | Globantis Labs`;
    54→  const description =
    55→    ind.intro ??
    56→    ind.heroHeading ??
    57→    ind.sections?.[0]?.paragraphs?.[0] ??
    58→    ind.tagline ??
    59→    ind.title;
    60→  return buildMetadata({
    61→    title,
    62→    description,
    63→    path: `/industries/${slug}`,
    64→    keywords: [ind.title, ind.tagline, ...(ind.focusAreas ?? [])].filter(
    65→      Boolean
    66→    ) as string[],
    67→  });
    68→}
